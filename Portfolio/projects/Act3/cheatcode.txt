$(document).ready(function(){
    $("#title").text("Good Morning!")

    $(this).css("background", "violet")
})
$("[href]").css("color", "blue")
$("ul li:first-child").css("color", "black")

$("th").css("background", "violet")

$("tr:even").css("background", "lightblue")

$("tr:odd").css("background", "lightpink")

$("#btn").click(function(){
    $("#message").text("You Clicked the button!")
})
$("#btn").dblclick(function(){
    $("#message").text("You double click the button!")
})
$("#btn").mouseenter(function(){
    $("#message").text("Your mouse reched the button")
})
$("#btn").mouseleave(function(){
    $("#message").text("Your mouse left the button")
})
$("#btn").mousedown(function(){
    $(this).css("background", "white")
})
$("#btn").mouseup(function(){
    $(this).css("background", "violet")
})
$("#name").focus(function(){
    $("#message").text("You select the text input")
    $(this).css("background", "lightyellow")
})
$("#name").blur(function(){
    $("#message").text("You leave the text input")
    $(this).css("background", "white")
})

$("course").change(function(){

})
$("#btn2").click(function(){
    $("#table").hide("slow")
})
